cmake_minimum_required(VERSION 3.16)
project(task_scheduler VERSION 0.4.0 LANGUAGES CXX)

# Embedding this project must not change the parent project's build policy.
set(SCHEDULER_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(SCHEDULER_STANDALONE ON)
endif()
option(SCHEDULER_BUILD_TESTS "Build scheduler tests" ${SCHEDULER_STANDALONE})
option(SCHEDULER_BUILD_EXAMPLES "Build scheduler examples" ${SCHEDULER_STANDALONE})
option(SCHEDULER_BUILD_BENCHMARKS "Build synthetic workload benchmarks" OFF)
option(SCHEDULER_THREAD_SANITIZER "Enable ThreadSanitizer" OFF)
option(SCHEDULER_ADDRESS_SANITIZER "Enable AddressSanitizer and UBSan" OFF)
if(SCHEDULER_THREAD_SANITIZER AND SCHEDULER_ADDRESS_SANITIZER)
  message(FATAL_ERROR "Use separate builds for thread and address sanitizers")
endif()
if((SCHEDULER_THREAD_SANITIZER OR SCHEDULER_ADDRESS_SANITIZER)
   AND NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "Sanitizer builds require GCC or Clang")
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
add_library(scheduler STATIC src/scheduler.cpp)
add_library(TaskScheduler::scheduler ALIAS scheduler)
target_compile_features(scheduler PUBLIC cxx_std_20)
set_target_properties(scheduler PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(scheduler PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(scheduler PUBLIC Threads::Threads)

function(scheduler_build_policy target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(SCHEDULER_THREAD_SANITIZER)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -g)
  elseif(SCHEDULER_ADDRESS_SANITIZER)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  endif()
endfunction()
scheduler_build_policy(scheduler)
# Static library consumers need the runtime when instrumentation is enabled.
if(SCHEDULER_THREAD_SANITIZER)
  target_link_options(scheduler PUBLIC -fsanitize=thread)
elseif(SCHEDULER_ADDRESS_SANITIZER)
  target_link_options(scheduler PUBLIC -fsanitize=address,undefined)
endif()

if(SCHEDULER_BUILD_TESTS)
  enable_testing()
  foreach(test scheduler_test scheduler_stress_test scheduler_lifecycle_test)
    add_executable(${test} tests/${test}.cpp)
    scheduler_build_policy(${test})
    if(MSVC)
      target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    target_link_libraries(${test} PRIVATE TaskScheduler::scheduler)
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 60)
  endforeach()
endif()
if(SCHEDULER_BUILD_EXAMPLES)
  add_executable(batch_example examples/batch.cpp)
  scheduler_build_policy(batch_example)
  target_link_libraries(batch_example PRIVATE TaskScheduler::scheduler)
  if(SCHEDULER_BUILD_TESTS)
    add_test(NAME batch_example COMMAND batch_example)
    set_tests_properties(batch_example PROPERTIES TIMEOUT 30)
  endif()
endif()

if(SCHEDULER_BUILD_BENCHMARKS)
  add_executable(scheduler_benchmark benchmarks/scheduler_benchmark.cpp)
  scheduler_build_policy(scheduler_benchmark)
  target_link_libraries(scheduler_benchmark PRIVATE TaskScheduler::scheduler)
  if(SCHEDULER_BUILD_TESTS)
    add_test(NAME benchmark_cpu_smoke COMMAND scheduler_benchmark cpu 2 16 200 10 0)
    add_test(NAME benchmark_blocking_smoke COMMAND scheduler_benchmark blocking 2 16 200 10 0)
    add_test(NAME benchmark_overload_smoke COMMAND scheduler_benchmark overload 2 16 64 0 4096)
    set_tests_properties(benchmark_cpu_smoke benchmark_blocking_smoke benchmark_overload_smoke
      PROPERTIES TIMEOUT 60)
  endif()
endif()

install(TARGETS scheduler EXPORT TaskSchedulerTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/scheduler.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(SCHEDULER_PACKAGE_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/TaskScheduler")
configure_package_config_file(cmake/TaskSchedulerConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/TaskSchedulerConfig.cmake"
  INSTALL_DESTINATION "${SCHEDULER_PACKAGE_DIR}")
write_basic_package_version_file(
  "${CMAKE_CURRENT_BINARY_DIR}/TaskSchedulerConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT TaskSchedulerTargets NAMESPACE TaskScheduler::
  DESTINATION "${SCHEDULER_PACKAGE_DIR}")
install(FILES
  "${CMAKE_CURRENT_BINARY_DIR}/TaskSchedulerConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/TaskSchedulerConfigVersion.cmake"
  DESTINATION "${SCHEDULER_PACKAGE_DIR}")
