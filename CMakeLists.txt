cmake_minimum_required(VERSION 3.16)
project(task_scheduler LANGUAGES CXX)
option(SCHEDULER_THREAD_SANITIZER "Build tests with ThreadSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(SCHEDULER_THREAD_SANITIZER)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(scheduler src/scheduler.cpp)
target_include_directories(scheduler PUBLIC include)
add_executable(scheduler_test tests/scheduler_test.cpp)
target_link_libraries(scheduler_test PRIVATE scheduler Threads::Threads)
add_executable(scheduler_stress_test tests/scheduler_stress_test.cpp)
target_link_libraries(scheduler_stress_test PRIVATE scheduler Threads::Threads)
enable_testing()
add_test(NAME scheduler_test COMMAND scheduler_test)
add_test(NAME scheduler_stress_test COMMAND scheduler_stress_test)
