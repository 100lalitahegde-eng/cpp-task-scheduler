cmake_minimum_required(VERSION 3.16)
project(scheduler_consumer LANGUAGES CXX)
if(SCHEDULER_SOURCE_DIR)
  add_subdirectory("${SCHEDULER_SOURCE_DIR}" scheduler-build)
  if(TARGET scheduler_test OR TARGET batch_example)
    message(FATAL_ERROR "Embedded scheduler enabled standalone targets")
  endif()
else()
  find_package(TaskScheduler 0.4 REQUIRED CONFIG)
endif()
get_target_property(consumer_options TaskScheduler::scheduler INTERFACE_COMPILE_OPTIONS)
if(consumer_options MATCHES "Werror|/WX")
  message(FATAL_ERROR "Library leaked its warning policy to the consumer")
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE TaskScheduler::scheduler)
enable_testing()
add_test(NAME consumer COMMAND consumer)
set_tests_properties(consumer PROPERTIES TIMEOUT 30)
